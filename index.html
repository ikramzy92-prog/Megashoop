-- ModernMarket.ge buyer orders/reviews foundation. Run in Supabase SQL Editor. -- This does not migrate old orders or connect the existing checkout. create table if not exists public.mm_buyer_orders ( id uuid primary key default gen_random_uuid(), buyer_id uuid not null references auth.users(id), status text not null default 'pending' check(status in ('pending','confirmed','shipped','delivered','cancelled')), delivery_address text, created_at timestamptz not null default now() ); create table if not exists public.mm_buyer_order_items ( id uuid primary key default gen_random_uuid(), order_id uuid not null references public.mm_buyer_orders(id) on delete cascade, product_id text not null, product_name text not null, quantity integer not null check(quantity>0), unit_price numeric(12,2) not null check(unit_price>=0) ); create table if not exists public.mm_product_reviews ( id uuid primary key default gen_random_uuid(), buyer_id uuid not null references auth.users(id), order_item_id uuid not null unique references public.mm_buyer_order_items(id), product_id text not null, rating smallint not null check(rating between 1 and 5), body text not null default '' check(char_length(body)<=2000), created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); create index if not exists mm_orders_buyer_idx on public.mm_buyer_orders(buyer_id,created_at desc); create index if not exists mm_items_order_idx on public.mm_buyer_order_items(order_id); create index if not exists mm_reviews_product_idx on public.mm_product_reviews(product_id); alter table public.mm_buyer_orders enable row level security; alter table public.mm_buyer_order_items enable row level security; alter table public.mm_product_reviews enable row level security; create policy "buyers read own orders" on public.mm_buyer_orders for select to authenticated using (buyer_id=auth.uid()); create policy "buyers read own order items" on public.mm_buyer_order_items for select to authenticated using (exists(select 1 from public.mm_buyer_orders o where o.id=order_id and o.buyer_id=auth.uid())); create policy "everyone reads reviews" on public.mm_product_reviews for select to anon,authenticated using(true); create policy "buyer reviews delivered purchases" on public.mm_product_reviews for insert to authenticated with check ( buyer_id=auth.uid() and exists ( select 1 from public.mm_buyer_order_items i join public.mm_buyer_orders o on o.id=i.order_id where i.id=order_item_id and i.product_id=mm_product_reviews.product_id and o.buyer_id=auth.uid() and o.status='delivered' )); -- No client INSERT/UPDATE policy for orders: a trusted server function must validate -- prices, quantities, stock and seller ownership before recording purchases. -- No public sales-count view: publish aggregated verified sales through a trusted backend.
